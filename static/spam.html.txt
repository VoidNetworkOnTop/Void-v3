    <div class="container">
        <h1>Tab Opener</h1>
        <p>Click the button below to start opening tabs and windows</p>
        <button id="startButton" class="button">Pp</button>
        <p class="warning">Warning: This will continue opening tabs until you close this tab!</p>
        <p id="status" class="status">Status: Idle</p>
    </div>

    <script>
        // Initialize variables
        let isRunning = false;
        let intervalId = null;
        const statusElement = document.getElementById('status');
        const startButton = document.getElementById('startButton');
        
        // Function to open a new tab or window
        function openNewTab() {
            // Open a new tab with about:blank
            window.open('about:blank', '_blank');
            
            // Also try to open a popup window
            window.open('about:blank', 'popup', 'width=400,height=400');
        }
        
        // Function to start the loop
        function startLoop() {
            if (!isRunning) {
                isRunning = true;
                statusElement.textContent = 'Status: Running - Opening tabs...';
                statusElement.style.color = '#ff0000';
                startButton.disabled = true;
                
                // Start the loop to open tabs every second
                intervalId = setInterval(openNewTab, 1000);
            }
        }
        
        // Add event listener to the button
        startButton.addEventListener('click', startLoop);
        
        // Warn user if they try to leave the page while the loop is running
        window.addEventListener('beforeunload', function(e) {
            if (isRunning) {
                // This message might not be displayed in modern browsers for security reasons
                e.returnValue = 'Tabs are still being opened. Are you sure you want to leave?';
            }
        });
    </script>
